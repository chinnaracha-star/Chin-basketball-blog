6. handleSubmit
function handleSubmit(event) {
  event.preventDefault();
  const nextErrors = {};
  if (!/^\S+@\S+\.\S+$/.test(values.email)) nextErrors.email = "Please enter a valid email address.";
  if (!values.password) nextErrors.password = "Please enter your password.";
  if (Object.keys(nextErrors).length === 0 && (values.email !== "demo@example.com" || values.password !== "password123")) {
    nextErrors.form = "Email or password is incorrect.";
  }
  setErrors(nextErrors);
  if (Object.keys(nextErrors).length === 0) {
    toast.success("Logged in successfully");
    navigate("/");
  }
}

ฟังก์ชันนี้ทำงานตอนผู้ใช้กดปุ่ม Submit หรือปุ่ม Log in

เพราะใน form มี

39. สรุปแบบภาพรวมง่าย ๆ

หน้า LoginPage นี้ทำงานแบบนี้:

ผู้ใช้พิมพ์ email/password
        ↓
handleChange อัปเดต values
        ↓
ผู้ใช้กด Log in
        ↓
handleSubmit ตรวจข้อมูล
        ↓
ถ้าผิด → setErrors → แสดง error
        ↓
ถ้าถูก → toast.success → navigate("/")

จำสั้น ๆ ได้ว่า:

values เก็บค่าที่พิมพ์
errors เก็บข้อความ error
handleChange ทำงานตอนพิมพ์
handleSubmit ทำงานตอนกด Log in
navigate("/") ใช้พาไปหน้าแรกหลัง Login สำเร็จ
