3. Function validate
function validate(values) {
  const errors = {};

  if (!values.name.trim()) errors.name = "Please enter your full name.";

  if (values.username.trim().length < 3) {
    errors.username = "Username must be at least 3 characters.";
  }

  if (!/^\S+@\S+\.\S+$/.test(values.email)) {
    errors.email = "Please enter a valid email address.";
  } else if (EXISTING_EMAILS.includes(values.email.trim().toLowerCase())) {
    errors.email = "This email is already in use.";
  }

  if (values.password.length < 8) {
    errors.password = "Password must be at least 8 characters.";
  }

  return errors;
}

10. Function handleSubmit
function handleSubmit(event) {
  event.preventDefault();

  const nextErrors = validate(values);

  setErrors(nextErrors);

  if (Object.keys(nextErrors).length === 0) {
    setIsSuccess(true);
  }
}

28. onChange
onChange={inputProps.onChange}

เมื่อผู้ใช้พิมพ์ จะเรียก Function ที่ได้รับมาจาก Props

ในที่นี้คือ

handleChange

จริง ๆ แล้ว onChange ถูกส่งเข้าไปผ่าน ...inputProps อยู่แล้ว ดังนั้นบรรทัดนี้ค่อนข้างซ้ำ

สามารถเขียนให้สั้นลงได้ว่า

<input
  id={inputId}
  {...inputProps}
  aria-invalid={Boolean(error)}
  aria-describedby={error ? errorId : undefined}
/>

เพราะ inputProps มี onChange อยู่แล้ว

29. aria-invalid
aria-invalid={Boolean(error)}

ใช้ช่วยเรื่อง Accessibility

บอก Screen Reader ว่า Input นี้มีข้อมูลไม่ถูกต้องหรือไม่

Boolean(error)

เปลี่ยนค่าให้เป็น Boolean

ถ้ามี Error

error = "Please enter a valid email address."

จะได้

true

ถ้าไม่มี Error

error = undefined

หรือ

error = ""

จะได้

false

ดังนั้นผลลัพธ์ใน HTML อาจเป็น

aria-invalid="true"

เมื่อข้อมูลผิด

30. aria-describedby
aria-describedby={error ? errorId : undefined}

ถ้ามี Error จะเชื่อม Input เข้ากับข้อความ Error

ตัวอย่าง

<input aria-describedby="email-error" />
<small id="email-error">
  Please enter a valid email address.
</small>

Screen Reader จะรู้ว่าข้อความ email-error เป็นคำอธิบายของ Input นี้

ถ้าไม่มี Error

aria-describedby={undefined}

React จะไม่ใส่ Attribute นี้ลงใน HTML

31. แสดงข้อความ Error
{error && (
  <small id={errorId} role="alert">
    {error}
  </small>

นี่คือ Conditional Rendering ด้วย &&

ความหมายคือ

ถ้ามี Error ให้แสดง <small>

ถ้า

error = "Password must be at least 8 characters."

จะแสดง

<small role="alert">
  Password must be at least 8 characters.
</small>

แต่ถ้า

error = ""

จะไม่แสดงอะไร

32. role="alert"
role="alert"

ช่วยให้ Screen Reader แจ้งข้อความ Error ทันทีเมื่อ Error ปรากฏขึ้น

values เป็นข้อความว่างทุกช่อง
errors เป็น Object ว่าง
isSuccess เป็น false
หน้าเว็บจึงแสดงฟอร์มสมัครสมาชิก

สมมติพิมพ์คำว่า

Chin
onChange ทำงาน
เรียก handleChange
ได้ name = "name"
ได้ value = "Chin"
อัปเดต values.name
ล้าง errors.name
React Render หน้าใหม่

State จะเป็น

{
  name: "Chin",
  username: "",
  email: "",
  password: ""
}
ตอนกดปุ่ม Sign up
Form เรียก handleSubmit
preventDefault() ป้องกันหน้า Refresh
ส่ง values ไปให้ validate
validate ตรวจ Name, Username, Email และ Password
ส่ง Object Error กลับมา
setErrors(nextErrors) เก็บ Error ลง State
ถ้ามี Error จะแสดง Error ใต้ Input
ถ้าไม่มี Error จะตั้ง isSuccess เป็น true
React แสดงหน้าสมัครสำเร็จแทนฟอร์ม
ตอนกด Continue
ปุ่มเรียก navigate("/login")
React Router เปลี่ยนไปหน้า Login
แนวคิดสำคัญที่ใช้ในหน้านี้

หน้า Signup นี้รวมความรู้ React หลายเรื่องไว้ด้วยกัน ได้แก่

useState เก็บข้อมูลที่เปลี่ยนแปลงได้
Controlled Component เชื่อม Input กับ State
Event Handling ด้วย onChange และ onSubmit
Form Validation ตรวจสอบข้อมูลก่อนสมัคร
Conditional Rendering เลือกแสดงฟอร์มหรือหน้าสำเร็จ
Reusable Component ใช้ AuthField ลดโค้ดซ้ำ
React Router ใช้ Link และ useNavigate
Accessibility ใช้ aria-invalid, aria-describedby และ role="alert"
